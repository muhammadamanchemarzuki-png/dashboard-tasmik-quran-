  • Paparan skrin penuh tanpa bar pelayar

Nota iPhone/iPad (Safari): Tekan ikon Share → pilih Add to Home Screen.

Langkah Demi Langkah Muat Naik ke GitHub Pages (Percuma)

  1. Sediakan Fail Utama: Klik butang Muat Turun index.html di atas (atau namakan fail HTML ini sebagai index.html).
  2. Cipta Repositori GitHub: Log masuk ke github.com, klik ikon + di penjuru kanan atas → pilih New repository.
  3. Tetapkan Nama Repositori: Taip nama repositori (contoh: dashboard-tasmik-quran), pilih Public, kemudian klik Create repository.
  4. Muat Naik Fail: Klik pautan uploading an existing file, seret fail index.html ke dalam ruangan muat naik, dan klik butang hijau Commit changes.
  5. Aktifkan GitHub Pages: Pergi ke tab Settings repositori → pilih menu Pages di sebelah kiri → di bawah Branch, tukar None kepada main (folder / (root)) → klik Save.
  6. Siap & Pasang PWA: Tunggu 1–2 minit dan segar semula halaman Pages untuk mendapatkan pautan rasmi anda (https://username.github.io/dashboard-tasmik-quran/). Buka pautan tersebut dan klik Pasang Aplikasi (PWA)!

Integrasi Google Sheets & Kod Google Apps Script

Sambungkan helaian SENARAI_MURID, REKOD_BACAAN & ANALISIS_BULANAN

Masukkan URL Web App Google Apps Script

Dashboard akan membaca helaian SENARAI_MURID, REKOD_BACAAN, dan ANALISIS_BULANAN serta menyimpan rekod baharu terus ke Google Sheets anda.

Salin keseluruhan kod di bawah dan tampal ke dalam editor Extensions > Apps Script di Google Sheets anda.

/**
 * GOOGLE APPS SCRIPT - API DASHBOARD BACAAN AL-QURAN & IQRA'
 * Menyokong bacaan data (doGet), tambah rekod (doPost), dan padam rekod (deleteRecord)
 */
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = {
      status: "success",
      timestamp: new Date().toISOString(),
      senarai_murid: getSheetData(ss, "SENARAI_MURID"),
      rekod_bacaan: getSheetData(ss, "REKOD_BACAAN"),
      analisis_bulanan: getSheetData(ss, "ANALISIS_BULANAN")
    };

    var jsonOutput = JSON.stringify(payload);
    if (e && e.parameter && e.parameter.callback) {
      return ContentService.createTextOutput(e.parameter.callback + "(" + jsonOutput + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(jsonOutput).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    var errJson = JSON.stringify({ status: "error", message: err.toString() });
    if (e && e.parameter && e.parameter.callback) {
      return ContentService.createTextOutput(e.parameter.callback + "(" + errJson + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(errJson).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("REKOD_BACAAN");
    if (!sheet) throw new Error("Helaian REKOD_BACAAN tidak ditemui.");

    var data = JSON.parse(e.postData.contents);

    // Tindakan 1: Padam Rekod Bacaan berdasarkan ID REKOD
    if (data.action === "deleteRecord") {
      var targetId = (data.recordId || "").toString().trim();
      if (!targetId) throw new Error("ID Rekod tidak sah.");
      var values = sheet.getDataRange().getDisplayValues();
      for (var i = values.length - 1; i >= 1; i--) {
        if ((values[i][0] || "").toString().trim() === targetId) {
          sheet.deleteRow(i + 1);
          return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Rekod berjaya dipadam" }))
            .setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "not_found", message: "ID Rekod tidak ditemui" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Tindakan 2: Tambah Rekod Bacaan Baharu
    var rec = data.record || data;
    sheet.appendRow([
      rec["ID REKOD"] || ("REC-" + new Date().getTime()),
      rec["TARIKH"] || "",
      rec["ID MURID"] || "",
      rec["NAMA MURID"] || "",
      rec["KELAS"] || "",
      rec["JENIS BACAAN"] || "",
      rec["BUKU IQRA"] || "-",
      rec["M/S MULA"] || "-",
      rec["M/S AKHIR"] || "-",
      rec["NAMA SURAH"] || "-",
      rec["AYAT MULA"] || "-",
      rec["AYAT AKHIR"] || "-",
      rec["KEHADIRAN TASMik"] || "HADIR",
      rec["KELANCARAN"] || "",
      rec["STATUS BACAAN"] || "",
      rec["CATATAN GURU"] || "",
      rec["DISEMAK OLEH"] || ""
    ]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Rekod berjaya disimpan" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var rows = sheet.getDataRange().getDisplayValues();
  if (rows.length < 2) return [];
  var headers = rows[0].map(function(h) { return h.toString().trim().toUpperCase(); });
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    var row = rows[i];
    if (row.join("").trim() === "") continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      if (headers[j]) obj[headers[j]] = row[j];
    }
    result.push(obj);
  }
  return result;
}
  1. Buka Google Sheets Asal Anda: Pastikan helaian bernama SENARAI_MURID, REKOD_BACAAN, dan ANALISIS_BULANAN wujud.
  2. Buka Editor Apps Script: Klik menu Extensions > Apps Script.
  3. Tampal Kod: Padam kod asal dalam Code.gs, tampal kod daripada tab 2. Kod Google Apps Script, dan tekan ikon Save (Ctrl+S).
  4. Terbitkan sebagai Web App: Klik Deploy > New deployment > Jenis: Web app > Execute as: Me > Who has access: Anyone > Klik Deploy.
  5. Salin URL Web App: Salin URL .../exec yang diberikan dan tampal ke dalam ruangan di tab 1. Sambung URL Web App.
